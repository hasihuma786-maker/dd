// doctors.html এ renderDoctors ফাংশন আপডেট করুন function renderDoctors(doctors) { if(!doctors || doctors.length === 0) { doctorsContainer.innerHTML = '
🔍 No doctors match your search.
'; return; } doctorsContainer.innerHTML = doctors.map(doctor => { // slug যদি NULL হয়, তাহলে নাম থেকে তৈরি করুন let doctorSlug = doctor.slug; if (!doctorSlug) { doctorSlug = doctor.name.toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''); } return `

${doctor.name}

${doctor.specialty}

⭐ ${doctor.rating || 4.5} • ${doctor.experience || 0}+ years

🏥 ${doctor.hospital || 'Private Practice'}

₹${doctor.fees || 500} per visit

View Profile →
`; }).join(''); }
